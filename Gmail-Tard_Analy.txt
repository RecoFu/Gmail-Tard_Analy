/**
 * Gmail 交易紀錄分析系統 - 分段執行版本
 * 解決 5 分鐘執行限制問題
 * 
 * 執行順序：
 * 1. step1_CollectEmails() - 收集郵件（可多次執行）
 * 2. step2_ParseData() - 解析資料
 * 3. step3_GenerateReports() - 生成報表
 */

// ========== 全域配置 ==========
const CONFIG = {
  LABEL: 'ordermaster',
  BATCH_SIZE: 100,  // 降低到 100 避免超時
  MAX_EXECUTION_TIME: 240000, // 4 分鐘限制
  SHEET_NAME_DETAIL: '交易明細(2012-2020)',
  SHEET_NAME_SUMMARY: '年度統計報表',
  SHEET_NAME_PROOF: '零知識證明',
  SHEET_NAME_RAW: '原始郵件暫存',
  START_YEAR: 2012,
  END_YEAR: 2020,
  PROP_PROGRESS: 'collection_progress',
  PROP_EMAIL_IDS: 'collected_email_ids'
};

// ========== 步驟1：收集郵件（支援多次執行）==========
function step1_CollectEmails() {
  const startTime = new Date().getTime();
  const props = PropertiesService.getScriptProperties();
  
  try {
    // 讀取上次進度
    let progress = props.getProperty(CONFIG.PROP_PROGRESS);
    let pageToken = null;
    let collectedIds = [];
    
    if (progress) {
      const data = JSON.parse(progress);
      pageToken = data.nextPageToken;
      const savedIds = props.getProperty(CONFIG.PROP_EMAIL_IDS);
      if (savedIds) {
        collectedIds = JSON.parse(savedIds);
      }
      Logger.log(`📌 繼續收集，已有 ${collectedIds.length} 封`);
    } else {
      Logger.log('🆕 開始新的收集任務');
    }
    
    const query = `label:${CONFIG.LABEL} after:${CONFIG.START_YEAR}/01/01 before:${CONFIG.END_YEAR}/12/31`;
    let batchCount = 0;
    let newCollected = 0;
    
    do {
      // 時間檢查
      if (new Date().getTime() - startTime > CONFIG.MAX_EXECUTION_TIME) {
        Logger.log('⏰ 接近時間限制，保存進度');
        props.setProperty(CONFIG.PROP_PROGRESS, JSON.stringify({
          nextPageToken: pageToken,
          timestamp: new Date().getTime()
        }));
        props.setProperty(CONFIG.PROP_EMAIL_IDS, JSON.stringify(collectedIds));
        
        Logger.log(`已收集 ${collectedIds.length} 封，請再次執行 step1_CollectEmails()`);
        
        try {
          SpreadsheetApp.getActiveSpreadsheet().toast(
            `已收集 ${collectedIds.length} 封，請再次執行 step1_CollectEmails()`,
            '⏸️ 暫停',
            10
          );
        } catch (e) {
          // 忽略 toast 錯誤
        }
        return;
      }
      
      // 搜尋郵件
      const response = Gmail.Users.Messages.list('me', {
        q: query,
        maxResults: CONFIG.BATCH_SIZE,
        pageToken: pageToken
      });
      
      if (response.messages) {
        batchCount++;
        Logger.log(`📧 批次 ${batchCount}: ${response.messages.length} 封`);
        
        // 只收集 ID，不下載完整內容
        response.messages.forEach(msg => {
          if (!collectedIds.includes(msg.id)) {
            collectedIds.push(msg.id);
            newCollected++;
          }
        });
        
        Logger.log(`✅ 累計: ${collectedIds.length} 封 (新增 ${newCollected})`);
      }
      
      pageToken = response.nextPageToken;
      
      // 每 5 批次保存一次
      if (batchCount % 5 === 0) {
        props.setProperty(CONFIG.PROP_EMAIL_IDS, JSON.stringify(collectedIds));
      }
      
    } while (pageToken);
    
    // 收集完成
    props.deleteProperty(CONFIG.PROP_PROGRESS);
    props.setProperty(CONFIG.PROP_EMAIL_IDS, JSON.stringify(collectedIds));
    
    // 按年份統計
    const yearCounts = countEmailsByYear(collectedIds);
    let summary = '📊 收集完成統計:\n\n';
    for (let year = CONFIG.START_YEAR; year <= CONFIG.END_YEAR; year++) {
      summary += `${year}: ${yearCounts[year] || 0} 封\n`;
    }
    summary += `\n總計: ${collectedIds.length} 封\n\n`;
    summary += '請執行 step2_ParseData() 解析資料';
    
    Logger.log(summary);
    
    try {
      SpreadsheetApp.getActiveSpreadsheet().toast(summary, '✅ 完成', 15);
    } catch (e) {
      // 獨立執行時 toast 不可用
      Logger.log('提示：請執行 step2_ParseData()');
    }
    
  } catch (error) {
    Logger.log(`❌ 錯誤: ${error.message}`);
    Logger.log(error.stack);
    
    try {
      SpreadsheetApp.getActiveSpreadsheet().toast(
        `錯誤: ${error.message}\n可再次執行繼續`,
        '❌ 失敗',
        10
      );
    } catch (e) {
      // 忽略 toast 錯誤
    }
  }
}

// ========== 步驟2：解析資料（分批處理）==========
function step2_ParseData() {
  const startTime = new Date().getTime();
  const props = PropertiesService.getScriptProperties();
  
  try {
    const idsJson = props.getProperty(CONFIG.PROP_EMAIL_IDS);
    if (!idsJson) {
      throw new Error('找不到郵件列表，請先執行 step1_CollectEmails()');
    }
    
    const allIds = JSON.parse(idsJson);
    Logger.log(`📋 準備解析 ${allIds.length} 封郵件`);
    
    // 讀取解析進度
    let parsedIndex = parseInt(props.getProperty('parse_index') || '0');
    
    // 取得 Spreadsheet
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) {
      throw new Error('無法取得試算表，請確認在 Google Sheets 中執行此腳本');
    }
    
    let sheet = ss.getSheetByName(CONFIG.SHEET_NAME_DETAIL);
    
    // 初始化工作表
    if (parsedIndex === 0) {
      if (!sheet) {
        sheet = ss.insertSheet(CONFIG.SHEET_NAME_DETAIL);
        Logger.log(`已建立工作表: ${CONFIG.SHEET_NAME_DETAIL}`);
      } else {
        sheet.clear();
        Logger.log(`已清空工作表: ${CONFIG.SHEET_NAME_DETAIL}`);
      }
      
      const headers = [
        '郵件ID', '郵件日期', '交易時間', '策略', '帳戶', '價格', 
        '交易方向', '數量', '倉位前', '倉位後', '系統狀態', '錯誤訊息', '原始內容'
      ];
      sheet.getRange(1, 1, 1, headers.length)
        .setValues([headers])
        .setFontWeight('bold')
        .setBackground('#4285F4')
        .setFontColor('#FFFFFF');
      
      Logger.log('已設定表頭');
    }
    
    const rows = [];
    let processed = 0;
    
    // 批次解析
    for (let i = parsedIndex; i < allIds.length; i++) {
      // 時間檢查
      if (new Date().getTime() - startTime > CONFIG.MAX_EXECUTION_TIME) {
        Logger.log(`⏰ 時間限制，已解析 ${i}/${allIds.length}`);
        
        // 寫入當前批次
        if (rows.length > 0) {
          sheet.getRange(sheet.getLastRow() + 1, 1, rows.length, 13).setValues(rows);
        }
        
        props.setProperty('parse_index', i.toString());
        
        Logger.log(`已解析 ${i}/${allIds.length} 封，請再次執行 step2_ParseData()`);
        
        try {
          SpreadsheetApp.getActiveSpreadsheet().toast(
            `已解析 ${i}/${allIds.length} 封\n請再次執行 step2_ParseData()`,
            '⏸️ 暫停',
            10
          );
        } catch (e) {
          // 忽略 toast 錯誤
        }
        return;
      }
      
      try {
        const msg = Gmail.Users.Messages.get('me', allIds[i], { format: 'full' });
        const record = parseEmailContent(msg);
        
        if (record) {
          rows.push([
            record.emailId,
            record.emailDate,
            record.tradeTime,
            record.strategy,
            record.account,
            record.price,
            record.direction,
            record.quantity,
            record.positionBefore,
            record.positionAfter,
            record.status,
            record.errorMsg,
            record.originalContent
          ]);
        }
        
        processed++;
        
        // 每 50 筆寫入一次
        if (rows.length >= 50) {
          sheet.getRange(sheet.getLastRow() + 1, 1, rows.length, 13).setValues(rows);
          rows.length = 0;
          Logger.log(`✅ 已寫入 ${i + 1}/${allIds.length}`);
        }
        
      } catch (error) {
        Logger.log(`⚠️ 郵件 ${allIds[i]} 解析失敗: ${error.message}`);
      }
      
      Utilities.sleep(50);
    }
    
    // 寫入剩餘資料
    if (rows.length > 0) {
      sheet.getRange(sheet.getLastRow() + 1, 1, rows.length, 13).setValues(rows);
    }
    
    sheet.autoResizeColumns(1, 13);
    
    // 清除進度
    props.deleteProperty('parse_index');
    
    Logger.log(`解析完成！共 ${processed} 筆，請執行 step3_GenerateReports()`);
    
    try {
      SpreadsheetApp.getActiveSpreadsheet().toast(
        `解析完成！\n請執行 step3_GenerateReports()`,
        '✅ 完成',
        10
      );
    } catch (e) {
      // 忽略 toast 錯誤
    }
    
  } catch (error) {
    Logger.log(`❌ 錯誤: ${error.message}`);
    Logger.log(error.stack);
    throw error;
  }
}

// ========== 步驟3：生成統計報表 ==========
function step3_GenerateReports() {
  try {
    Logger.log('開始生成報表...');
    
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) {
      throw new Error('無法取得試算表，請確認在 Google Sheets 中執行此腳本');
    }
    
    const detailSheet = ss.getSheetByName(CONFIG.SHEET_NAME_DETAIL);
    
    if (!detailSheet) {
      throw new Error('找不到明細資料，請先執行 step2_ParseData()');
    }
    
    // 讀取所有明細資料
    const data = detailSheet.getDataRange().getValues();
    const headers = data[0];
    const rows = data.slice(1);
    
    Logger.log(`讀取 ${rows.length} 筆明細`);
    
    // 轉換為物件陣列
    const parsedData = rows.map(row => ({
      emailId: row[0],
      emailDate: row[1],
      tradeTime: row[2],
      strategy: row[3],
      account: row[4],
      price: row[5],
      direction: row[6],
      quantity: row[7],
      positionBefore: row[8],
      positionAfter: row[9],
      status: row[10],
      errorMsg: row[11],
      originalContent: row[12]
    }));
    
    // 生成統計報表（包含零知識證明）
    generateSummaryReportV2(parsedData);
    
    Logger.log('所有報表生成完成！');
    
    try {
      SpreadsheetApp.getActiveSpreadsheet().toast(
        '所有報表已完成！\n請檢視各工作表',
        '完成',
        10
      );
    } catch (e) {
      // 忽略 toast 錯誤
    }
    
  } catch (error) {
    Logger.log(`錯誤: ${error.message}`);
    Logger.log(error.stack);
    throw error;
  }
}

// ========== 輔助函數 ==========
function countEmailsByYear(emailIds) {
  const yearCounts = {};
  
  // 初始化所有年份
  for (let year = CONFIG.START_YEAR; year <= CONFIG.END_YEAR; year++) {
    yearCounts[year] = 0;
  }
  
  // 抽樣統計（前 200 封）
  const sampleSize = Math.min(200, emailIds.length);
  
  Logger.log(`正在統計年份分布（抽樣 ${sampleSize} 封）...`);
  
  for (let i = 0; i < sampleSize; i++) {
    try {
      const msg = Gmail.Users.Messages.get('me', emailIds[i], { format: 'minimal' });
      const year = new Date(parseInt(msg.internalDate)).getFullYear();
      
      if (year >= CONFIG.START_YEAR && year <= CONFIG.END_YEAR) {
        yearCounts[year] = (yearCounts[year] || 0) + 1;
      }
      
      // 每 50 封輸出一次進度
      if ((i + 1) % 50 === 0) {
        Logger.log(`  統計進度: ${i + 1}/${sampleSize}`);
      }
      
    } catch (e) {
      Logger.log(`  跳過郵件 ${emailIds[i]}: ${e.message}`);
    }
    
    Utilities.sleep(100); // 避免 API 限流
  }
  
  // 按比例推估全部
  const ratio = emailIds.length / sampleSize;
  const estimatedCounts = {};
  
  for (let year = CONFIG.START_YEAR; year <= CONFIG.END_YEAR; year++) {
    estimatedCounts[year] = Math.round(yearCounts[year] * ratio);
  }
  
  Logger.log('年份分布統計完成（推估值）');
  
  return estimatedCounts;
}

// ========== 郵件解析（完整保留）==========
function parseEmailContent(message) {
  const headers = message.payload.headers;
  const emailId = message.id;
  const date = new Date(parseInt(message.internalDate));
  
  const subject = headers.find(h => h.name === 'Subject')?.value || '';
  let body = '';
  
  try {
    // 處理 body.data
    if (message.payload.body && message.payload.body.data) {
      const bodyData = message.payload.body.data;
      
      if (typeof bodyData === 'string') {
        body = Utilities.newBlob(
          Utilities.base64Decode(bodyData.replace(/-/g, '+').replace(/_/g, '/'))
        ).getDataAsString();
      }
    } 
    
    // 處理 parts（multipart 郵件）
    if (!body && message.payload.parts) {
      for (const part of message.payload.parts) {
        if (part.mimeType === 'text/plain' && part.body && part.body.data) {
          const partData = part.body.data;
          
          if (typeof partData === 'string') {
            body += Utilities.newBlob(
              Utilities.base64Decode(partData.replace(/-/g, '+').replace(/_/g, '/'))
            ).getDataAsString();
          }
        }
      }
    }
    
    // 備用：使用 snippet
    if (!body && message.snippet) {
      body = message.snippet;
    }
    
    // 解碼 HTML 實體
    body = body.replace(/&quot;/g, '"')
               .replace(/&amp;/g, '&')
               .replace(/&lt;/g, '<')
               .replace(/&gt;/g, '>')
               .replace(/&#39;/g, "'");
    
  } catch (e) {
    Logger.log(`郵件 ${emailId} 內容提取失敗: ${e.message}`);
    body = (message.snippet || '').replace(/&quot;/g, '"');
  }
  
  const year = date.getFullYear();
  
  // 2012-2018 使用舊格式（包含2015-2018的混合格式）
  if (year >= 2012 && year <= 2018) {
    return parseFormat2012to2014(emailId, date, subject, body);
  } 
  // 2019-2020 才使用新格式
  else if (year >= 2019 && year <= 2020) {
    return parseFormat2015to2020(emailId, date, subject, body);
  }
  
  return null;
}

function parseFormat2012to2014(emailId, date, subject, body) {
  const record = {
    emailId: emailId,
    emailDate: Utilities.formatDate(date, 'GMT+8', 'yyyy/MM/dd HH:mm:ss'),
    tradeTime: '',
    strategy: '',
    account: '',
    price: '',
    direction: '',
    quantity: 0,
    positionBefore: 0,
    positionAfter: 0,
    status: 'normal',
    errorMsg: '',
    originalContent: body.substring(0, 200)
  };
  
  // 檢查是否為錯誤訊息或系統通知
  if (body.includes('錯誤') || body.includes('失敗') || body.includes('取消') || 
      body.includes('開始自動轉倉') || body.includes('API 錯誤')) {
    record.status = 'error';
    record.errorMsg = body.trim().substring(0, 200);
    const timeMatch = body.match(/(\d{4}\/\d{2}\/\d{2}\s+\d{2}:\d{2}:\d{2})/);
    if (timeMatch) record.tradeTime = timeMatch[1];
    return record;
  }
  
  // 提取策略名稱（支援多種格式）
  const strategyMatch = body.match(/\n([A-Z]{2,4})\s*$/m) || 
                        subject.match(/OrderMaster-([A-Z]+)/) ||
                        body.match(/([A-Z]{2,4})$/);
  if (strategyMatch) record.strategy = strategyMatch[1];
  
  // 提取時間
  const timeMatch = body.match(/(\d{4}\/\d{2}\/\d{2}\s+\d{2}:\d{2}:\d{2})/);
  if (timeMatch) record.tradeTime = timeMatch[1];
  
  // 提取價格（支援小數點）
  const priceMatch = body.match(/"(\d+\.?\d*)"/);
  if (priceMatch) record.price = priceMatch[1];
  
  // 提取交易方向
  if (body.includes('多單平倉') || body.includes('平多')) {
    record.direction = 'close_long';
  } else if (body.includes('空單平倉') || body.includes('平空')) {
    record.direction = 'close_short';
  } else if (body.includes('多翻空') || body.includes('long_to_short')) {
    record.direction = 'long_to_short';
  } else if (body.includes('空翻多') || body.includes('short_to_long')) {
    record.direction = 'short_to_long';
  } else if (body.includes('加碼多單') || body.includes('做多') || body.includes('開多')) {
    record.direction = 'add_long';
  } else if (body.includes('加碼空單') || body.includes('做空') || body.includes('開空')) {
    record.direction = 'add_short';
  } else if (body.includes('減碼多單')) {
    record.direction = 'reduce_long';
  } else if (body.includes('減碼空單')) {
    record.direction = 'reduce_short';
  }
  
  // 提取倉位資訊（處理+號和無+號兩種格式）
  const posMatch = body.match(/目前倉位[：:]\s*([+\-]?\d+)/);
  if (posMatch) {
    record.positionAfter = parseInt(posMatch[1].replace('+', ''));
  }
  
  const prevPosMatch = body.match(/之前倉位[：:]\s*([+\-]?\d+)/);
  if (prevPosMatch) {
    record.positionBefore = parseInt(prevPosMatch[1].replace('+', ''));
  }
  
  // 計算數量
  record.quantity = record.positionAfter - record.positionBefore;
  
  // 提取帳戶（支援多種券商）
  if (body.includes('元大')) record.account = '元大';
  else if (body.includes('凱基') || body.includes('KGI')) record.account = 'KGI';
  else if (body.includes('群益')) record.account = '群益';
  
  return record;
}

function parseFormat2015to2020(emailId, date, subject, body) {
  const record = {
    emailId: emailId,
    emailDate: Utilities.formatDate(date, 'GMT+8', 'yyyy/MM/dd HH:mm:ss'),
    tradeTime: '',
    strategy: '',
    account: '',
    price: '',
    direction: '',
    quantity: 0,
    positionBefore: 0,
    positionAfter: 0,
    status: 'normal',
    errorMsg: '',
    originalContent: body.substring(0, 200)
  };
  
  // 檢查是否為錯誤訊息或系統通知
  if (body.includes('API 錯誤') || body.includes('連線失敗') || body.includes('錯誤') ||
      body.includes('開始自動轉倉') || body.includes('轉倉完成')) {
    record.status = 'error';
    record.errorMsg = body.trim().substring(0, 200);
    const timeMatch = body.match(/(\d{4}\/\d{2}\/\d{2}\s+\d{2}:\d{2}:\d{2})/);
    if (timeMatch) record.tradeTime = timeMatch[1];
    return record;
  }
  
  // 提取策略【TCA】或從主旨提取
  const strategyMatch = body.match(/【([A-Z]+)】/) || 
                        subject.match(/OrderMaster-([A-Z]+)/);
  if (strategyMatch) record.strategy = strategyMatch[1];
  
  // 提取時間
  const timeMatch = body.match(/(\d{4}\/\d{2}\/\d{2}\s+\d{2}:\d{2}:\d{2})/);
  if (timeMatch) record.tradeTime = timeMatch[1];
  
  // 提取價格（支援小數點）
  const priceMatch = body.match(/"(\d+\.?\d*)"/);
  if (priceMatch) record.price = priceMatch[1];
  
  // 提取交易方向
  if (body.includes('加碼多單')) {
    record.direction = 'add_long';
  } else if (body.includes('減碼多單')) {
    record.direction = 'reduce_long';
  } else if (body.includes('加碼空單')) {
    record.direction = 'add_short';
  } else if (body.includes('減碼空單')) {
    record.direction = 'reduce_short';
  } else if (body.includes('多單平倉') || body.includes('平多')) {
    record.direction = 'close_long';
  } else if (body.includes('空單平倉') || body.includes('平空')) {
    record.direction = 'close_short';
  } else if (body.includes('多翻空')) {
    record.direction = 'long_to_short';
  } else if (body.includes('空翻多')) {
    record.direction = 'short_to_long';
  }
  
  // 提取倉位（兩種格式）
  // 格式1: "策略倉位 9 → 10"
  const positionMatch = body.match(/策略倉位\s+([+\-]?\d+)\s*→\s*([+\-]?\d+)/);
  if (positionMatch) {
    record.positionBefore = parseInt(positionMatch[1].replace('+', ''));
    record.positionAfter = parseInt(positionMatch[2].replace('+', ''));
  }
  
  // 格式2: "目前倉位：+2 之前倉位：1"（2015早期格式）
  if (!positionMatch) {
    const posMatch = body.match(/目前倉位[：:]\s*([+\-]?\d+)/);
    if (posMatch) {
      record.positionAfter = parseInt(posMatch[1].replace('+', ''));
    }
    
    const prevPosMatch = body.match(/之前倉位[：:]\s*([+\-]?\d+)/);
    if (prevPosMatch) {
      record.positionBefore = parseInt(prevPosMatch[1].replace('+', ''));
    }
  }
  
  // 提取帳戶【KGI】或從文字提取
  const accountMatch = body.match(/帳戶【([A-Z]+)】/) ||
                       body.match(/([A-Z]{2,4})\([^)]+\)/);
  if (accountMatch) {
    record.account = accountMatch[1];
  } else if (body.includes('KGI')) {
    record.account = 'KGI';
  } else if (body.includes('元大')) {
    record.account = '元大';
  } else if (body.includes('凱基')) {
    record.account = 'KGI';
  }
  
  // 計算數量
  record.quantity = record.positionAfter - record.positionBefore;
  
  return record;
}

function calculateYearStatsV2(trades, year) {
  let yearTrades = trades;
  if (year !== null) {
    yearTrades = trades.filter(t => {
      if (!t.tradeTime) return false;
      const tradeYear = new Date(t.tradeTime).getFullYear();
      return tradeYear === year;
    });
  }
  
  if (yearTrades.length === 0) {
    return {
      totalTrades: 0, totalPnL: 0, totalROI: 0, annualizedROI: 0,
      cagr: 0, winRate: 0, profitFactor: 0, avgProfit: 0, avgLoss: 0,
      maxDrawdown: 0, strategyReturn: 0, benchmarkReturn: 8.0,
      excessReturn: -8.0, opportunityCost: 0, recommendation: 'N/A'
    };
  }
  
  const POINT_VALUE = 200;
  let totalPnL = 0, wins = 0, losses = 0, totalProfit = 0, totalLoss = 0;
  const pnlHistory = [0];
  let currentPosition = 0, entryPrice = 0;
  
  for (const trade of yearTrades) {
    if (!trade.price || trade.quantity === 0) continue;
    
    const price = parseFloat(trade.price);
    const qty = trade.quantity;
    
    if (currentPosition === 0) {
      currentPosition = qty;
      entryPrice = price;
    } else if ((currentPosition > 0 && qty < 0) || (currentPosition < 0 && qty > 0)) {
      const closedQty = Math.min(Math.abs(currentPosition), Math.abs(qty));
      const priceDiff = currentPosition > 0 ? (price - entryPrice) : (entryPrice - price);
      const pnl = priceDiff * closedQty * POINT_VALUE;
      
      totalPnL += pnl;
      pnlHistory.push(totalPnL);
      
      if (pnl > 0) {
        wins++;
        totalProfit += pnl;
      } else if (pnl < 0) {
        losses++;
        totalLoss += Math.abs(pnl);
      }
      
      currentPosition += qty;
      if (currentPosition !== 0) entryPrice = price;
    } else {
      currentPosition += qty;
    }
  }
  
  const totalTrades = wins + losses;
  const winRate = totalTrades > 0 ? (wins / totalTrades) * 100 : 0;
  const profitFactor = totalLoss > 0 ? totalProfit / totalLoss : (totalProfit > 0 ? 999 : 0);
  const avgProfit = wins > 0 ? totalProfit / wins : 0;
  const avgLoss = losses > 0 ? totalLoss / losses : 0;
  
  let maxDrawdown = 0, peak = 0;
  for (const pnl of pnlHistory) {
    if (pnl > peak) peak = pnl;
    if (peak > 0) {
      const dd = ((peak - pnl) / peak) * 100;
      if (dd > maxDrawdown) maxDrawdown = dd;
    }
  }
  
  const INITIAL_CAPITAL = 1000000;
  const totalROI = (totalPnL / INITIAL_CAPITAL) * 100;
  const years = year ? 1 : 9;
  const annualizedROI = totalROI / years;
  const finalValue = INITIAL_CAPITAL + totalPnL;
  const cagr = finalValue > 0 ? (Math.pow(finalValue / INITIAL_CAPITAL, 1 / years) - 1) * 100 : -100;
  
  let recommendation = '';
  if (totalPnL > 0 && winRate > 50 && profitFactor > 1.5) {
    recommendation = '優秀策略，建議持續';
  } else if (totalPnL > 0 && winRate > 45) {
    recommendation = '穩定獲利';
  } else if (totalPnL < 0) {
    recommendation = '需優化策略';
  } else {
    recommendation = '持續監控';
  }
  
  return {
    totalTrades, 
    totalPnL: Math.round(totalPnL),
    totalROI: parseFloat(totalROI.toFixed(2)),
    annualizedROI: parseFloat(annualizedROI.toFixed(2)),
    cagr: parseFloat(cagr.toFixed(2)),
    winRate: parseFloat(winRate.toFixed(2)),
    profitFactor: parseFloat(profitFactor.toFixed(2)),
    avgProfit: Math.round(avgProfit),
    avgLoss: Math.round(avgLoss),
    maxDrawdown: parseFloat(maxDrawdown.toFixed(2)),
    strategyReturn: parseFloat(annualizedROI.toFixed(2)),
    benchmarkReturn: 8.0,
    excessReturn: parseFloat((annualizedROI - 8.0).toFixed(2)),
    opportunityCost: Math.round(8.0 * years * INITIAL_CAPITAL / 100),
    recommendation
  };
}

function diagnoseEmails() {
  Logger.log('========== 郵件診斷 ==========');
  
  const query = `label:${CONFIG.LABEL}`;
  const response = Gmail.Users.Messages.list('me', {
    q: query,
    maxResults: 3
  });
  
  if (!response.messages) {
    Logger.log('未找到郵件');
    return;
  }
  
  Logger.log(`找到 ${response.messages.length} 封樣本\n`);
  
  for (let i = 0; i < response.messages.length; i++) {
    const msg = Gmail.Users.Messages.get('me', response.messages[i].id, { format: 'full' });
    const date = new Date(parseInt(msg.internalDate));
    const subject = msg.payload.headers.find(h => h.name === 'Subject')?.value || '';
    
    Logger.log(`--- 郵件 ${i + 1} ---`);
    Logger.log(`ID: ${msg.id}`);
    Logger.log(`日期: ${date.toLocaleString()}`);
    Logger.log(`主旨: ${subject}`);
    Logger.log(`Snippet: ${msg.snippet}`);
    
    const parsed = parseEmailContent(msg);
    Logger.log(`解析: ${JSON.stringify(parsed, null, 2)}\n`);
  }
}

    Logger.log(`  狀態: ${row[10]}`);

  // 統計狀態
  let normalCount = 0;
  let errorCount = 0;
  let withPrice = 0;
  let withQuantity = 0;
  
  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    if (row[10] === 'normal') normalCount++;
    if (row[10] === 'error') errorCount++;

  }

  Logger.log(`  正常: ${normalCount}`);
  Logger.log(`  錯誤: ${errorCount}`);
