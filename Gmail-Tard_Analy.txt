function step1_CollectEmails() {
  const startTime = new Date().getTime();
  const props = PropertiesService.getScriptProperties();
  
  try {
    let progress = props.getProperty('collection_progress');
    let pageToken = null;
    let collectedIds = [];
    
    if (progress) {
      const data = JSON.parse(progress);
      pageToken = data.nextPageToken;
      const savedIds = props.getProperty(CONFIG.PROP_EMAIL_IDS);
      if (savedIds) collectedIds = JSON.parse(savedIds);
    }
    
    const query = `label:${CONFIG.LABEL} after:2012/01/01 before:2020/12/31`;
    let batchCount = 0;
    
    do {
      if (new Date().getTime() - startTime > CONFIG.MAX_EXECUTION_TIME) {
        props.setProperty('collection_progress', JSON.stringify({nextPageToken: pageToken}));
        props.setProperty(CONFIG.PROP_EMAIL_IDS, JSON.stringify(collectedIds));
        Logger.log(`已收集 ${collectedIds.length} 封`);
        return;
      }
      
      const response = Gmail.Users.Messages.list('me', {q: query, maxResults: CONFIG.BATCH_SIZE, pageToken: pageToken});
      
      if (response.messages) {
        batchCount++;
        response.messages.forEach(msg => {
          if (!collectedIds.includes(msg.id)) collectedIds.push(msg.id);
        });
      }
      
      pageToken = response.nextPageToken;
      
    } while (pageToken);
    
    props.deleteProperty('collection_progress');
    props.setProperty(CONFIG.PROP_EMAIL_IDS, JSON.stringify(collectedIds));
    Logger.log(`收集完成：${collectedIds.length} 封`);
    
  } catch (e) {
    Logger.log(`錯誤: ${e.message}`);
  }
}

function step2_ParseData() {
  const startTime = new Date().getTime();
  const props = PropertiesService.getScriptProperties();
  
  try {
    const idsJson = props.getProperty(CONFIG.PROP_EMAIL_IDS);
    if (!idsJson) throw new Error('找不到郵件列表');
    
    const allIds = JSON.parse(idsJson);
    let parsedIndex = parseInt(props.getProperty('parse_index') || '0');
    
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName(CONFIG.SHEET_NAME_DETAIL);
    
    if (parsedIndex === 0) {
      if (!sheet) sheet = ss.insertSheet(CONFIG.SHEET_NAME_DETAIL);
      else sheet.clear();
      
      sheet.getRange(1, 1, 1, 13).setValues([[
        '郵件ID', '郵件日期', '交易時間', '策略', '帳戶', '價格',
        '交易方向', '數量', '倉位前', '倉位後', '系統狀態', '錯誤訊息', '原始內容'
      ]]).setFontWeight('bold').setBackground('#4285F4').setFontColor('#FFFFFF');
    }
    
    const rows = [];
    
    for (let i = parsedIndex; i < allIds.length; i++) {
      if (new Date().getTime() - startTime > CONFIG.MAX_EXECUTION_TIME) {
        if (rows.length > 0) sheet.getRange(sheet.getLastRow() + 1, 1, rows.length, 13).setValues(rows);
        props.setProperty('parse_index', i.toString());
        return;
      }
      
      try {
        const msg = Gmail.Users.Messages.get('me', allIds[i], {format: 'full'});
        const record = parseEmail(msg);
        if (record) {
          rows.push([
            record.emailId, record.emailDate, record.tradeTime, record.strategy,
            record.account, record.price, record.direction, record.quantity,
            record.positionBefore, record.positionAfter, record.status,
            record.errorMsg, record.originalContent
          ]);
        }
        
        if (rows.length >= 50) {
          sheet.getRange(sheet.getLastRow() + 1, 1, rows.length, 13).setValues(rows);
          rows.length = 0;
        }
      } catch (e) {}
      
      Utilities.sleep(50);
    }
    
    if (rows.length > 0) sheet.getRange(sheet.getLastRow() + 1, 1, rows.length, 13).setValues(rows);
    sheet.autoResizeColumns(1, 13);
    props.deleteProperty('parse_index');
    
  } catch (e) {
    Logger.log(`錯誤: ${e.message}`);
  }
}

function step3_GenerateReports() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const detailSheet = ss.getSheetByName(CONFIG.SHEET_NAME_DETAIL);
    if (!detailSheet) throw new Error('找不到明細資料');
    
    const data = detailSheet.getDataRange().getValues();
    const rows = data.slice(1);
    
    const parsedData = rows.map(row => ({
      emailId: row[0], emailDate: row[1], tradeTime: row[2], strategy: row[3],
      account: row[4], price: row[5], direction: row[6], quantity: row[7],
      positionBefore: row[8], positionAfter: row[9], status: row[10],
      errorMsg: row[11], originalContent: row[12]
    }));
    
    generateActualReport(parsedData);
    
  } catch (e) {
    Logger.log(`錯誤: ${e.message}`);
  }
}

function generateActualReport(data) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(CONFIG.SHEET_NAME_SUMMARY);
  if (!sheet) sheet = ss.insertSheet(CONFIG.SHEET_NAME_SUMMARY);
  else sheet.clear();
  
  const headers = [
    '年度', '交易次數', '錯誤次數', '期初權益', '外部入金',
    '期末權益', '交易損益', '年度報酬(%)', '累計報酬(%)', '投資建議'
  ];
  
  sheet.getRange(1, 1, 1, 10).setValues([headers])
    .setFontWeight('bold').setBackground('#0F9D58').setFontColor('#FFFFFF');
  
  const allTrades = data.filter(d => d.tradeTime)
    .sort((a, b) => new Date(a.tradeTime) - new Date(b.tradeTime));
  
  const rows = [];
  
  // 2012: 元大帳戶起始
  rows.push([
    2012, countTrades(allTrades, 2012), countErrors(allTrades, 2012),
    0, 50000, 50000, 0, 0, 0, '起始'
  ]);
  
  // 2013: 推算（假設小幅獲利）
  const est2013 = 80000;
  rows.push([
    2013, countTrades(allTrades, 2013), countErrors(allTrades, 2013),
    50000, 0, est2013, 30000, 60, 60, '推估獲利'
  ]);
  
  // 2014: 推算（假設持續獲利）
  const est2014 = 150000;
  rows.push([
    2014, countTrades(allTrades, 2014), countErrors(allTrades, 2014),
    est2013, 0, est2014, 70000, 87.5, 200, '推估獲利'
  ]);
  
  // 2015: 累積至360K後轉出
  rows.push([
    2015, countTrades(allTrades, 2015), countErrors(allTrades, 2015),
    est2014, 0, 360000, 210000, 140, 620, '大幅獲利'
  ]);
  
  // 2016: 轉入KGI（360K → 315K，虧損）
  rows.push([
    2016, countTrades(allTrades, 2016), countErrors(allTrades, 2016),
    360000, 0, 315493, -44507, -12.36, 530.99, '換券商虧損'
  ]);
  
  // 2017: 翻倍
  rows.push([
    2017, countTrades(allTrades, 2017), countErrors(allTrades, 2017),
    315493, 0, 637711, 322218, 102.14, 1175.42, '優秀'
  ]);
  
  // 2018: 追加入金360K
  const start2018 = 637711;
  const deposit2018 = 360000;
  const end2018 = 997678;
  const pnl2018 = end2018 - start2018 - deposit2018;
  const return2018 = (pnl2018 / start2018) * 100;
  
  rows.push([
    2018, countTrades(allTrades, 2018), countErrors(allTrades, 2018),
    start2018, deposit2018, end2018, pnl2018, parseFloat(return2018.toFixed(2)),
    parseFloat(((end2018 - 410000) / 410000 * 100).toFixed(2)), '持平'
  ]);
  
  // 2019: 峰值135萬（對帳單顯示「前日餘額94,904」是某個交易日）
  const start2019 = 997678;
  const peak2019 = 1350000;
  const pnl2019 = peak2019 - start2019;
  const return2019 = (pnl2019 / start2019) * 100;
  
  rows.push([
    2019, countTrades(allTrades, 2019), countErrors(allTrades, 2019),
    start2019, 0, peak2019, pnl2019, parseFloat(return2019.toFixed(2)),
    parseFloat(((peak2019 - 410000) / 410000 * 100).toFixed(2)), '達峰值'
  ]);
  
  // 2020: 崩盤
  const start2020 = 1350000;
  const end2020 = 30000;
  const pnl2020 = end2020 - start2020;
  const return2020 = (pnl2020 / start2020) * 100;
  
  rows.push([
    2020, countTrades(allTrades, 2020), countErrors(allTrades, 2020),
    start2020, 0, end2020, pnl2020, parseFloat(return2020.toFixed(2)),
    parseFloat(((end2020 - 410000) / 410000 * 100).toFixed(2)), '斷頭'
  ]);
  
  // 總計
  const totalTrades = rows.reduce((sum, r) => sum + (typeof r[1] === 'number' ? r[1] : 0), 0);
  const totalErrors = rows.reduce((sum, r) => sum + (typeof r[2] === 'number' ? r[2] : 0), 0);
  const totalDeposits = 410000;
  const totalPnL = 30000 - 410000;
  const totalReturn = (totalPnL / totalDeposits) * 100;
  
  rows.push([
    '總計', totalTrades, totalErrors, '-', totalDeposits, 30000,
    totalPnL, '-', parseFloat(totalReturn.toFixed(2)), '累計虧損'
  ]);
  
  sheet.getRange(2, 1, rows.length, 10).setValues(rows);
  
  // 格式化
  sheet.getRange(2, 4, rows.length, 3).setNumberFormat('#,##0');
  sheet.getRange(2, 7, rows.length, 1).setNumberFormat('#,##0');
  sheet.getRange(2, 8, rows.length, 2).setNumberFormat('0.00');
  
  sheet.getRange(rows.length + 1, 1, 1, 10).setFontWeight('bold').setBackground('#FFF9C4');
  
  // 條件格式

    SpreadsheetApp.newConditionalFormatRule().whenNumberGreaterThan(0)
      .setBackground('#D4EDDA').setRanges([pnlRange]).build(),
    SpreadsheetApp.newConditionalFormatRule().whenNumberLessThan(0)
      .setBackground('#F8D7DA').setRanges([pnlRange]).build()
  ]);

function countTrades(trades, year) {
  return trades.filter(t => {
    if (t.status !== 'normal') return false;
    const d = new Date(t.tradeTime);
    return d.getFullYear() === year;
  }).length;
}

function countErrors(trades, year) {
  return trades.filter(t => {
    if (t.status !== 'error') return false;
    if (!t.tradeTime) return false;
    const d = new Date(t.tradeTime);
    return d.getFullYear() === year;
  }).length;
}

function parseEmail(msg) {
  const headers = msg.payload.headers;
  const id = msg.id;
  const date = new Date(parseInt(msg.internalDate));
  const subj = headers.find(h => h.name === 'Subject')?.value || '';
  
  let body = '';
  try {
    if (msg.payload.body && msg.payload.body.data) {
      const bd = msg.payload.body.data;
      if (typeof bd === 'string') {
        body = Utilities.newBlob(Utilities.base64Decode(bd.replace(/-/g, '+').replace(/_/g, '/'))).getDataAsString();
      }
    }
    
    if (!body && msg.payload.parts) {
      for (const part of msg.payload.parts) {
        if (part.mimeType === 'text/plain' && part.body && part.body.data) {
          const pd = part.body.data;
          if (typeof pd === 'string') {
            body += Utilities.newBlob(Utilities.base64Decode(pd.replace(/-/g, '+').replace(/_/g, '/'))).getDataAsString();
          }
        }
      }
    }
    
    if (!body && msg.snippet) body = msg.snippet;
    body = body.replace(/&quot;/g, '"').replace(/&amp;/g, '&');
    
  } catch (e) {
    body = (msg.snippet || '').replace(/&quot;/g, '"');
  }
  
  return parseTrade(id, date, subj, body);
}

function parseTrade(id, date, subj, body) {
  const rec = {
    emailId: id, emailDate: Utilities.formatDate(date, 'GMT+8', 'yyyy/MM/dd HH:mm:ss'),
    tradeTime: '', strategy: '', account: '', price: '', direction: '',
    quantity: 0, positionBefore: 0, positionAfter: 0, status: 'normal',
    errorMsg: '', originalContent: body.substring(0, 200)
  };
  
  if (body.includes('錯誤') || body.includes('失敗') || body.includes('轉倉')) {
    rec.status = 'error';
    rec.errorMsg = body.trim().substring(0, 200);
    const tm = body.match(/(\d{4}\/\d{2}\/\d{2}\s+\d{2}:\d{2}:\d{2})/);
    if (tm) rec.tradeTime = tm[1];
    return rec;
  }
  
  const strat = body.match(/【([A-Z]+)】/) || body.match(/\n([A-Z]{2,4})\s*$/m) || subj.match(/OrderMaster-([A-Z]+)/);
  if (strat) rec.strategy = strat[1];
  
  const tm = body.match(/(\d{4}\/\d{2}\/\d{2}\s+\d{2}:\d{2}:\d{2})/);
  if (tm) rec.tradeTime = tm[1];
  
  const pr = body.match(/"(\d+\.?\d*)"/);
  if (pr) rec.price = pr[1];
  
  if (body.includes('加碼多單')) rec.direction = 'add_long';
  else if (body.includes('減碼多單')) rec.direction = 'reduce_long';
  else if (body.includes('多單平倉')) rec.direction = 'close_long';
  else if (body.includes('空翻多')) rec.direction = 'short_to_long';
  else if (body.includes('多翻空')) rec.direction = 'long_to_short';
  
  const pos = body.match(/策略倉位\s+([+\-]?\d+)\s*→\s*([+\-]?\d+)/);
  if (pos) {
    rec.positionBefore = parseInt(pos[1].replace('+', ''));
    rec.positionAfter = parseInt(pos[2].replace('+', ''));
  } else {
    const curr = body.match(/目前倉位[：:]\s*([+\-]?\d+)/);
    const prev = body.match(/之前倉位[：:]\s*([+\-]?\d+)/);
    if (curr) rec.positionAfter = parseInt(curr[1].replace('+', ''));
    if (prev) rec.positionBefore = parseInt(prev[1].replace('+', ''));
  }
  
  rec.quantity = rec.positionAfter - rec.positionBefore;
  
  if (body.includes('KGI') || body.includes('凱基')) rec.account = 'KGI';
  else if (body.includes('元大')) rec.account = '元大';
  
  return rec;
}
